# 3) 独立查看 MuJoCo 场景（需 X11 转发，本地终端跑）：
python -c "
import mujoco, mujoco.viewer, time
m = mujoco.MjModel.from_xml_path('/root/gpufree-data/RLinf/assets/frankasim_xmls/arena.xml')
d = mujoco.MjData(m)
mujoco.mj_forward(m, d)
with mujoco.viewer.launch_passive(m, d) as v:
    while v.is_running():
        v.sync()
        time.sleep(0.01)
"

# 4) 离屏渲染录 mp4（无 X11，服务器上跑）：
python -c "
import mujoco, numpy as np, imageio
m = mujoco.MjModel.from_xml_path('/root/gpufree-data/RLinf/assets/frankasim_xmls/arena.xml')
d = mujoco.MjData(m)
r = mujoco.Renderer(m, 480, 640)
frames = []
for i in range(200):
    mujoco.mj_step(m, d)
    r.update_scene(d)
    frames.append(r.render())
imageio.mimsave('/tmp/arena.mp4', frames, fps=30)
print('saved /tmp/arena.mp4')
"
